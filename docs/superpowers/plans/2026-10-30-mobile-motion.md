# Mobile Motion System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the mobile app a Premium motion system — generated motion tokens, a `src/core/motion` vocabulary on reanimated, three shared primitives, and the confirmed increment-1 surfaces (buttons, list entrances, bottom sheet, tab bar, skeletons).

**Architecture:** `brand/tokens.json` gains a `motion.native` block that `brand/build.mjs` generates into `mobile/src/core/theme/tokens/motion.ts`. A hand-written `src/core/motion/motion.ts` is the only module importing `react-native-reanimated`; it exports timings, springs, the `fadeRise` entrance, stagger helpers and a reduced-motion hook. Shared primitives (`PressableScale`, `MotionEntrance`, `Skeleton`) consume it; feature components consume primitives only.

**Tech Stack:** Expo SDK 57 / RN 0.86, `react-native-reanimated` v4 (SDK-matched), jest-expo + React Testing Library, brand codegen (`npm run brand`).

**Spec:** `docs/superpowers/specs/2026-10-30-mobile-motion-design.md`

- [ ] **Step 3: Implement `mobile/src/core/motion/motion.ts`**

```ts
import { useEffect, useState } from 'react';
import { Easing, FadeIn, FadeInDown, useReducedMotion as useReanimatedReducedMotion } from 'react-native-reanimated';
import { motionDurations, motionEasing, motionSprings, motionStagger } from '../theme/tokens/motion';

/** The Premium motion vocabulary. Every duration, easing and spring derives from the generated tokens —
 *  components consume this module (or the shared primitives) and never import reanimated directly. */

export const useReducedMotion = useReanimatedReducedMotion;

const bezier = (points: readonly number[]) => Easing.bezier(points[0], points[1], points[2], points[3]);

/** Timed moves: entrances and on-screen changes use the signature decelerate curve, exits accelerate out. */
export const timings = {
  quick: { duration: motionDurations.quick, easing: bezier(motionEasing.standard) },
  base: { duration: motionDurations.base, easing: bezier(motionEasing.standard) },
  enter: { duration: motionDurations.enter, easing: bezier(motionEasing.standard) },
  reveal: { duration: motionDurations.reveal, easing: bezier(motionEasing.standard) },
  exit: { duration: motionDurations.slow, easing: bezier(motionEasing.exit) },
} as const;

/** Springs are for interactions only — entrances use `timings`. */
export const springs = motionSprings;

/** The one entrance style: fade + 12px rise over `enter`, staggered by index. Reduced motion collapses
 *  it to a plain quick fade with no travel. */
export function fadeRise(delayMs: number, reduced: boolean) {
  if (reduced) return FadeIn.duration(motionDurations.quick);
  return FadeInDown.duration(motionDurations.enter).easing(bezier(motionEasing.standard)).delay(delayMs);
}

/** Micro-cascade: index × step, capped at maxItems so the total stays inside the 500ms budget. */
export function staggerDelay(index: number): number {
  return index < motionStagger.maxItems ? index * motionStagger.stepMs : 0;
}

/** How long the first-batch entrance window stays open: enter + full stagger, with headroom. */
export const ENTRANCE_WINDOW_MS = motionDurations.enter + motionStagger.maxItems * motionStagger.stepMs + 150;

/** True only while the list's first batch may animate. The first render with items opens a one-shot
 *  window; once it closes, rows mounted later (pagination, scroll-back recycling) render instantly
 *  instead of replaying the entrance mid-scroll. */
export function useEntranceWindow(hasItems: boolean): boolean {
  const [armed, setArmed] = useState(true);
  useEffect(() => {
    if (!hasItems) return;
    const timer = setTimeout(() => setArmed(false), ENTRANCE_WINDOW_MS);
    return () => clearTimeout(timer);
  }, [hasItems]);
  return armed;
}
```

```tsx
import { Platform, Pressable, type PressableProps, type PressableStateCallbackType } from 'react-native';
import Animated, { useAnimatedStyle, useSharedValue, withSpring } from 'react-native-reanimated';
import { springs, useReducedMotion } from '../../core/motion/motion';
import { useTheme } from '../../core/theme/theme-provider';

const PRESSED_OPACITY = 0.6;
const PRESSED_SCALE = 0.97;

const AnimatedPressable = Animated.createAnimatedComponent(Pressable);

export interface PressableFeedbackProps extends PressableProps {
  /** Off for surfaces that aren't controls, e.g. a sheet's backdrop, so tapping it doesn't flash the screen. */
  feedback?: boolean;
  /** Adds the Premium press dip (scale 0.97, spring) on top of the dim/ripple. Off by default. */
  scale?: boolean;
}

/** Every tappable thing in the app: it answers the finger at once (a ripple on Android, a dim on iOS, an
 *  optional spring dip), so a tap never feels missed while the work behind it runs. Plain Pressable shows
 *  nothing on press. */
export function PressableFeedback({ feedback = true, scale = false, style, android_ripple, ...rest }: PressableFeedbackProps) {
  const theme = useTheme();
  const reduced = useReducedMotion();
  const responds = !rest.disabled && !!(rest.onPress || rest.onLongPress);
  const dims = feedback && responds;
  const dips = scale && responds && !reduced; // reduced motion keeps the dim/ripple, drops the dip
  const dip = useSharedValue(1);
  const dipStyle = useAnimatedStyle(() => ({ transform: [{ scale: dip.value }] }));
  const ripple = dims && Platform.OS === 'android'
    ? android_ripple ?? { color: theme.colors.pressedOverlay, foreground: true }
    : android_ripple;
  const pressedStyle = (state: PressableStateCallbackType) => {
    const base = typeof style === 'function' ? style(state) : style;
    return dims && state.pressed && Platform.OS !== 'android' ? [base, { opacity: PRESSED_OPACITY }] : base;
  };
  return (
    <AnimatedPressable
      {...rest}
      android_ripple={ripple}
      style={(state: PressableStateCallbackType) => [pressedStyle(state), dipStyle]}
      onPressIn={(e) => {
        if (dips) dip.value = withSpring(PRESSED_SCALE, springs.press);
        rest.onPressIn?.(e);
      }}
      onPressOut={(e) => {
        if (dips) dip.value = withSpring(1, springs.press);
        rest.onPressOut?.(e);
      }}
    />
  );
}
```

```tsx
import Animated, { useAnimatedStyle, useSharedValue, withTiming } from 'react-native-reanimated';
import { timings, useReducedMotion } from '../../core/motion/motion';
import { useTheme } from '../../core/theme/theme-provider';
import { PressableFeedback } from './PressableFeedback';
import { AppText } from './AppText';

interface Props {
  label: string;
  onPress: () => void;
  variant?: 'primary' | 'secondary' | 'ghost';
  loading?: boolean;
  disabled?: boolean;
  /** Overrides the label for screen readers when several buttons share one, e.g. "Accept Ananya Rao". */
  accessibilityLabel?: string;
}

export function AppButton({ label, onPress, variant = 'primary', loading = false, disabled = false, accessibilityLabel }: Props) {
  const theme = useTheme();
  const reduced = useReducedMotion();
  const inactive = disabled || loading;
  const isPrimary = variant === 'primary';
  const isSecondary = variant === 'secondary'; // white pill with a hairline border, e.g. "I already have an account"
  const pressed = useSharedValue(0);
  // The pressed surface eases in over `quick`; reduced motion flips it instantly.
  const surface = useAnimatedStyle(() => {
    const target = isPrimary
      ? pressed.value ? theme.colors.brandPressed : theme.colors.brand
      : isSecondary ? theme.colors.surface : 'transparent';
    return { backgroundColor: withTiming(target, { duration: reduced ? 0 : timings.quick.duration, easing: timings.base.easing }) };
  });
  return (
    <PressableFeedback
      feedback={false}
      scale
      accessibilityRole="button"
      accessibilityLabel={accessibilityLabel}
      accessibilityState={{ disabled: inactive, busy: loading }}
      onPress={inactive ? undefined : onPress}
      onPressIn={() => { pressed.value = 1; }}
      onPressOut={() => { pressed.value = 0; }}
      style={(state) => [
        {
          height: 52,
          borderRadius: theme.radii.pill,
          alignItems: 'center',
          justifyContent: 'center',
          borderColor: isPrimary ? 'transparent' : isSecondary ? theme.colors.separator : theme.colors.brand,
          borderWidth: isPrimary ? 0 : 1,
          opacity: inactive ? 0.4 : state.pressed && !isPrimary ? 0.85 : 1,
        },
        surface,
        isPrimary ? theme.shadows.md : {},
      ]}
    >
      {loading ? (
        <ActivityIndicator testID="app-button-spinner" color={isPrimary ? theme.colors.textOnBrand : theme.colors.brand} />
      ) : (
        <AppText variant="button" style={{ color: isPrimary ? theme.colors.textOnBrand : isSecondary ? theme.colors.textPrimary : theme.colors.brand }}>
          {label}
        </AppText>
      )}
    </PressableFeedback>
  );
}
```

```tsx
import type { ReactNode } from 'react';
import Animated from 'react-native-reanimated';
import { fadeRise, staggerDelay, useReducedMotion } from '../../core/motion/motion';

interface MotionEntranceProps {
  /** Position in the list — drives the stagger (capped at 8 items). */
  index: number;
  /** False once useEntranceWindow has closed: the item renders without an entrance. */
  armed: boolean;
  children: ReactNode;
}

/** The one entrance pattern (fade + rise, staggered by index). Wrap list items with it and arm it from
 *  `useEntranceWindow` so recycled rows never replay the animation mid-scroll. */
export function MotionEntrance({ index, armed, children }: MotionEntranceProps) {
  const reduced = useReducedMotion();
  return (
    <Animated.View entering={armed ? fadeRise(staggerDelay(index), reduced) : undefined}>
      {children}
    </Animated.View>
  );
}
```

- [ ] **Step 4: Create `Skeleton.tsx`**

```tsx
import { useEffect } from 'react';
import { StyleSheet, View, type StyleProp, type ViewStyle } from 'react-native';
import Animated, { cancelAnimation, Easing, useAnimatedStyle, useSharedValue, withRepeat, withTiming } from 'react-native-reanimated';
import { useReducedMotion } from '../../core/motion/motion';
import { useTheme } from '../../core/theme/theme-provider';

const SWEEP_MS = 1200;

interface SkeletonProps {
  height?: number;
  width?: number | `${number}%`;
  radius?: number;
  style?: StyleProp<ViewStyle>;
}

/** A first-load placeholder block: a highlight band sweeping through a soft surface (the one sanctioned
 *  linear loop — loading). Reduced motion keeps the band still. Compose blocks into rows; no spinner. */
export function Skeleton({ height = 96, width, radius, style }: SkeletonProps) {
  const theme = useTheme();
  const reduced = useReducedMotion();
  const sweep = useSharedValue(0);
  useEffect(() => {
    if (reduced) return;
    sweep.value = withRepeat(withTiming(1, { duration: SWEEP_MS, easing: Easing.linear }), -1, true);
    return () => cancelAnimation(sweep);
  }, [reduced, sweep]);
  const band = useAnimatedStyle(() => ({ transform: [{ translateX: `${sweep.value * 160 - 30}%` }] }));
  return (
    <View
      accessibilityLabel="Loading"
      style={[{ height, width, borderRadius: radius ?? theme.radii.card, backgroundColor: theme.colors.canvas, overflow: 'hidden' }, style]}
    >
      <Animated.View style={[StyleSheet.absoluteFill, { backgroundColor: theme.colors.surface, opacity: 0.7 }, band]} />
    </View>
  );
}
```

```tsx
import { useEffect, useRef, type ReactNode } from 'react';
import { KeyboardAvoidingView, Modal, Platform, ScrollView, View, useWindowDimensions } from 'react-native';
import Animated, { useAnimatedStyle, useSharedValue, withSpring, withTiming } from 'react-native-reanimated';
import { springs, timings, useReducedMotion } from '../../core/motion/motion';
import { PressableFeedback } from './PressableFeedback';
import { AppText } from './AppText';
import { useTheme } from '../../core/theme/theme-provider';

interface BottomSheetProps {
  visible: boolean;
  title: string;
  onClose: () => void;
  children: ReactNode;
  /** Pinned below the scrolling content, e.g. the submit button. */
  footer?: ReactNode;
}

/** The Atelier bottom sheet: a scrim that closes it, a rounded white panel. The panel springs up, and on
 *  a manual close it eases back down before unmounting — the Modal's own fade covers the scrim both ways. */
export function BottomSheet({ visible, title, onClose, children, footer }: BottomSheetProps) {
  const theme = useTheme();
  const insets = useSafeAreaInsets();
  const { height } = useWindowDimensions();
  const reduced = useReducedMotion();
  const offscreen = useSharedValue(1); // 1 = below the screen, 0 = in place
  const closing = useRef(false);
  useEffect(() => {
    if (!visible) {
      offscreen.value = 1;
      closing.current = false;
      return;
    }
    offscreen.value = reduced ? 0 : withSpring(0, springs.settle);
  }, [visible, offscreen, reduced]);
  const close = () => {
    if (closing.current) return;
    closing.current = true;
    offscreen.value = withTiming(1, timings.exit);
    setTimeout(onClose, reduced ? 0 : timings.exit.duration); // unmount after the exit lands
  };
  const panel = useAnimatedStyle(() => ({ transform: [{ translateY: offscreen.value * height }] }));
  return (
    <Modal visible={visible} transparent animationType="fade" onRequestClose={close}>
      <View style={{ flex: 1, backgroundColor: theme.colors.overlay }}>
        <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : undefined} style={{ flex: 1 }}>
          <PressableFeedback feedback={false} accessibilityLabel="Close" style={{ flex: 1 }} onPress={close} />
          <Animated.View style={[{ maxHeight: '85%', backgroundColor: theme.colors.surface, borderTopLeftRadius: theme.radii.sheet, borderTopRightRadius: theme.radii.sheet, paddingBottom: insets.bottom + theme.spacing[4] }, panel]}>
            <ScrollView contentContainerStyle={{ padding: theme.spacing[6], paddingBottom: theme.spacing[2], gap: theme.spacing[3.5] }} keyboardShouldPersistTaps="handled">
              <AppText variant="title" style={{ color: theme.colors.heading }}>{title}</AppText>
              {children}
            </ScrollView>
            {footer ? <View style={{ paddingHorizontal: theme.spacing[6] }}>{footer}</View> : null}
          </Animated.View>
        </KeyboardAvoidingView>
      </View>
    </Modal>
  );
}
```

```tsx
import Svg, { Circle, Path, Rect } from 'react-native-svg';
import { type ColorValue } from 'react-native';
import Animated, { useAnimatedStyle, useSharedValue, withSpring } from 'react-native-reanimated';
import { springs, useReducedMotion } from '../../core/motion/motion';

export type TabIconName = 'home' | 'explore' | 'bookings' | 'profile' | 'calendar' | 'messages' | 'feed' | 'shop';

/** Atelier tab bar glyphs (24px line icons from the prototype). */
function Glyph({ name, color }: { name: TabIconName; color: ColorValue }): ReactElement {
  const common = { width: 24, height: 24, viewBox: '0 0 24 24', fill: 'none', stroke: color, strokeWidth: 1.8, strokeLinecap: 'round', strokeLinejoin: 'round' } as const;
  switch (name) {

      return <Svg {...common}><Path d="M3 10l9-7 9 7v10a1 1 0 0 1-1 1h-5v-6H9v6H4a1 1 0 0 1-1-1z" /></Svg>;
    case 'explore':

    case 'bookings':

    case 'calendar':
      return <Svg {...common}><Rect x={3} y={5} width={18} height={16} rx={3} /><Path d="M3 10h18M8 3v4M16 3v4" /></Svg>;
    case 'shop':
      return <Svg {...common}><Path d="M5 8h14l-1.2 12H6.2z" /><Path d="M9 8V6.5a3 3 0 0 1 6 0V8" /></Svg>;
    case 'feed':
      return <Svg {...common}><Rect x={3} y={3} width={18} height={18} rx={5} /><Path d="M10 8.5v7l6-3.5z" /></Svg>;
    case 'messages':
      return <Svg {...common}><Path d="M21 12a8 8 0 0 1-11.5 7.2L4 21l1.8-5.5A8 8 0 1 1 21 12z" /></Svg>;
    case 'profile':
      return <Svg {...common}><Circle cx={12} cy={8} r={4} /><Path d="M4 21a8 8 0 0 1 16 0" /></Svg>;
  }

/** Selecting a tab settles the glyph from a slight dip back to full size — a quiet pulse, no bounce
 *  (Premium: 0% overshoot). Reduced motion skips the dip. */
export function TabIcon({ name, color, focused = false }: { name: TabIconName; color: ColorValue; focused?: boolean }) {
  const reduced = useReducedMotion();
  const scale = useSharedValue(1);
  useEffect(() => {
    if (!focused || reduced) {
      scale.value = 1;
      return;
    }
    scale.value = 0.88;
    scale.value = withSpring(1, springs.press);
  }, [focused, reduced, scale]);
  const settle = useAnimatedStyle(() => ({ transform: [{ scale: scale.value }] }));
  return (
    <Animated.View style={settle}>
      <Glyph name={name} color={color} />
    </Animated.View>
  );
}
```
